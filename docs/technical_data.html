3. API nội bộ (pseudocode)

Combat Manager

// Khởi tạo combat với đội hình hero và enemy list
void StartCombat(List<Hero> heroes, List<Enemy> enemies);

// Xác định thứ tự lượt dựa trên speed (Combatant.speed + random)
List<Combatant> DetermineTurnOrder();

// Xử lý một lượt của combatant (hero hoặc enemy)
void ProcessTurn(Combatant current);

// Khi hero chọn skill
void UseSkill(Hero user, Skill skill, List<Combatant> targets);

// Tính toán sát thương (xem game_systems.html § 6)
int CalculateDamage(Hero attacker, Enemy target, Skill skill, bool isCrit);

// Áp dụng effect (stun, poison, riposte, heal, stress...)
void ApplyEffect(Combatant target, Effect effect);

// Kiểm tra kết thúc combat (mọi hero hoặc mọi enemy có HP <= 0)
bool IsCombatEnded();

Stress & Affliction System

// Thêm stress (clamp 0..100)
void AddStress(Hero hero, int amount);

// Khi stress >= 100 → trigger resolve
void ResolveStress(Hero hero);

// 70% chance Affliction, 30% chance Virtue
Affliction SelectRandomAffliction();
Virtue     SelectRandomVirtue();

// Áp dụng / hủy
void ApplyAffliction(Hero hero, Affliction aff);
void RemoveAffliction(Hero hero);

Dungeon Navigation

// Tạo dungeon từ stage data + seed
Dungeon GenerateDungeon(StageConfig stage, long seed);

// Hiển thị các lựa chọn pathway tại node hiện tại
List<PathNode> GetCurrentChoices();

// Khi người chơi chọn một nhánh
void ChoosePath(PathNode chosenNode);

// Xử lý node hiện tại (combat, event, rest, reward...)
void ProcessCurrentNode();

// Chuyển sang node tiếp theo
void AdvanceToNextNode();

Event Handler

void TriggerEvent(EventData eventData);
void ShowEventUI(EventData eventData);
void ApplyEventOutcome(List<Outcome> outcomes);
Outcome PickRandomOutcome(List<Outcome> outcomes);

Hamlet Manager

void InitHamlet();
void UpgradeBuilding(BuildingType type);
bool CureDisease(Hero hero, Disease disease);
void ReduceStressInTavern(Hero hero);
Hero RecruitHero(string className);

Save/Load

void SaveGame(string slotName);
void LoadGame(string slotName);

struct SaveData {
    string version;
    int gold;
    List<HeroSave> heroes;
    List<BuildingSave> buildings;
    int currentStage;
    int dungeonProgress;
}

4. Flowcharts

Combat Turn Flow

Start Turn │ ▼ Determine Turn Order (by speed) │ ▼ ┌─→ Has combatant left? ──No──→ End Combat │ │ Yes │ ▼ │ Combatant type? │ ├─ Hero → Show skill menu → Player chooses → Calculate dmg & effects ─┐ │ │ │ │ └─ Enemy → AI picks skill → Use on target ───────────────────────────┤ │ │ └──────────────── Apply damage, stress, effects ◄──────────────────────────┘ │ ▼ Remove dead combatants │ └───── loop ─────►

Stress Resolution Flow

Add Stress │ ▼ Stress >= 100? ──No──→ Return │ Yes ▼ Resolve → random(0,1) │ ├── ≤ 0.7 → Pick random Affliction → Apply modifiers ──┐ │ │ └── > 0.7 → Pick random Virtue → Apply buffs ────┤ │ Set stress to 0 ◄─────────────┘

Dungeon Pathway Flow

Enter Dungeon → Load stage_N.json │ ▼ ┌─→ Current node index < total? ──No──→ Fight boss → End stage │ │ Yes │ ▼ │ Get current node │ │ │ ▼ │ Node type? │ ├─ Combat → Start combat → Win? → rewards + advance │ ├─ Event → Trigger event → outcomes → advance │ ├─ Rest → Choose hero → stress reduce → advance │ ├─ Reward → Roll drop table → give items → advance │ └─ Choice → Show 2-3 branches → player picks → next layer │ │ └──────────────────────────────────────────────────────────►─┘

Event Processing Flow

Trigger Event → Load event data │ ▼ Show description + choices │ ▼ Player selects choice │ ▼ For each outcome in choice: │ ▼ random check → Pass → Apply outcome effect → Fail → Skip │ ▼ Update UI + game state │ ▼ Continue dungeon

Hamlet Management Flow

Enter Hamlet → Show main UI │ ▼ Player action? ├─ Upgrade building → Check gold + level → Upgrade ├─ Cure disease → Select hero + disease → Pay gold → Remove ├─ Reduce stress → Select hero → Pay gold → -stress ├─ Recruit hero → Generate random hero → Add to roster └─ Start next stage → Load dungeon
